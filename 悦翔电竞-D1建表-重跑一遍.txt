DROP TABLE IF EXISTS users;
CREATE TABLE IF NOT EXISTS users ( uid TEXT PRIMARY KEY, username TEXT NOT NULL UNIQUE, password TEXT NOT NULL, tag TEXT DEFAULT 'deop', op_level INTEGER DEFAULT 1, status TEXT DEFAULT 'yes', created_at TEXT );
CREATE TABLE IF NOT EXISTS reports ( id INTEGER PRIMARY KEY AUTOINCREMENT, reporter_name TEXT, target_id TEXT, reason TEXT, submit_time TEXT, status TEXT DEFAULT 'pending', reject_reason TEXT DEFAULT '' );
CREATE TABLE IF NOT EXISTS orders ( id INTEGER PRIMARY KEY AUTOINCREMENT, booster TEXT DEFAULT '', uid TEXT, username TEXT, product_title TEXT, game_id TEXT, screenshot_key TEXT, order_no TEXT, server_area TEXT, designated_booster TEXT DEFAULT '', message TEXT, submit_time TEXT, status TEXT DEFAULT 'pending', reject_reason TEXT DEFAULT '' );
CREATE INDEX IF NOT EXISTS idx_reports_time ON reports (submit_time DESC);
CREATE INDEX IF NOT EXISTS idx_orders_time ON orders (submit_time DESC);
CREATE INDEX IF NOT EXISTS idx_orders_uid ON orders (uid);
CREATE INDEX IF NOT EXISTS idx_orders_booster ON orders (booster);
CREATE TABLE IF NOT EXISTS products ( id INTEGER PRIMARY KEY AUTOINCREMENT, title TEXT, price TEXT, description TEXT, created_at TEXT );
CREATE TABLE IF NOT EXISTS booster_info ( id INTEGER PRIMARY KEY AUTOINCREMENT, uid TEXT DEFAULT '', name TEXT, rank TEXT, specialty TEXT, intro TEXT );
CREATE TABLE IF NOT EXISTS reviews ( id INTEGER PRIMARY KEY AUTOINCREMENT, order_id INTEGER, reviewer_name TEXT, reviewer_uid TEXT, booster_name TEXT, booster_uid TEXT, stars INTEGER, comment TEXT, review_time TEXT );
CREATE TABLE IF NOT EXISTS admin_log ( id INTEGER PRIMARY KEY AUTOINCREMENT, admin_uid TEXT, admin_name TEXT, action TEXT, log_time TEXT );
CREATE INDEX IF NOT EXISTS idx_reviews_order ON reviews (order_id);
CREATE INDEX IF NOT EXISTS idx_reviews_booster ON reviews (booster_uid);
CREATE INDEX IF NOT EXISTS idx_admin_log_time ON admin_log (log_time DESC);
